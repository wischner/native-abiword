# Fyodor on any Native backend. `make` remains the local entry point; this
# file is what the platform SDKs build.
cmake_minimum_required(VERSION 3.20)
project(fyodor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(FYODOR_VERSION "0.1.0" CACHE STRING "Application version")
if(NOT NATIVE_ROOT)
    # Same GitHub clone the Makefile uses; -DNATIVE_ROOT=/path uses a checkout.
    execute_process(COMMAND sh "${CMAKE_SOURCE_DIR}/scripts/build/fetch_native.sh"
        OUTPUT_VARIABLE NATIVE_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE
        COMMAND_ERROR_IS_FATAL ANY)
endif()
set(BUILD_TESTING OFF CACHE BOOL "Native standalone tests" FORCE)
add_subdirectory(src/ui native-dependency)

set(libraries impexp edit fmt gfx doc util)
add_library(fyodor_options INTERFACE)
target_include_directories(fyodor_options INTERFACE include)
foreach(name ${libraries})
    target_include_directories(fyodor_options INTERFACE "lib/${name}/include")
endforeach()
target_compile_options(fyodor_options INTERFACE
    $<$<CXX_COMPILER_ID:GNU>:-Wall;-Wextra;-pedantic>)
target_compile_definitions(fyodor_options INTERFACE
    FYODOR_VERSION="${FYODOR_VERSION}")
# Native's probe found std::jthread only behind libc++'s experimental
# library; its add_compile_options does not reach this directory.
if(APPLE AND NOT NATIVE_HAS_CXX20_THREADS)
    target_compile_options(fyodor_options INTERFACE -fexperimental-library)
    target_link_options(fyodor_options INTERFACE -fexperimental-library)
endif()

set(core_sources)
foreach(name ${libraries})
    file(GLOB sources CONFIGURE_DEPENDS "lib/${name}/*.cpp")
    list(APPEND core_sources ${sources})
endforeach()
add_library(fyodor_core STATIC ${core_sources})
target_link_libraries(fyodor_core PUBLIC fyodor_options)

file(GLOB ui_sources CONFIGURE_DEPENDS src/ui/*.cpp src/app/*.cpp)
# Cross SDKs consume host-generated C++; never execute a target-built rc.
set(FYODOR_RESOURCE_CPP "" CACHE FILEPATH "Resource C++ generated by a host Native rc")
if(FYODOR_RESOURCE_CPP)
    if(NOT EXISTS "${FYODOR_RESOURCE_CPP}")
        message(FATAL_ERROR "FYODOR_RESOURCE_CPP does not exist")
    endif()
    set(resource_cpp "${FYODOR_RESOURCE_CPP}")
elseif(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Cross builds require FYODOR_RESOURCE_CPP; use scripts/build/platforms.py or generate it with make resources")
else()
    set(resource_cpp "${CMAKE_BINARY_DIR}/resources/fyodor.cpp")
    file(GLOB resource_images CONFIGURE_DEPENDS src/resources/icons/*/*.png)
    add_custom_command(OUTPUT "${resource_cpp}"
        BYPRODUCTS "${CMAKE_BINARY_DIR}/resources/fyodor.resources"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/resources"
        COMMAND $<TARGET_FILE:native_rc>
            "${CMAKE_SOURCE_DIR}/src/resources/fyodor.json"
            -o "${CMAKE_BINARY_DIR}/resources/fyodor.resources"
            --cpp "${resource_cpp}"
        DEPENDS native_rc src/resources/fyodor.json ${resource_images}
        VERBATIM)
endif()
add_custom_target(fyodor_resources DEPENDS "${resource_cpp}")
add_library(fyodor_ui STATIC ${ui_sources} "${resource_cpp}")
target_include_directories(fyodor_ui PUBLIC src/ui)
target_link_libraries(fyodor_ui PUBLIC fyodor_core native)

add_executable(fyodor WIN32 MACOSX_BUNDLE src/main.cpp)
target_link_libraries(fyodor PRIVATE fyodor_ui)
if(APPLE)
    set_target_properties(fyodor PROPERTIES
        MACOSX_BUNDLE_BUNDLE_NAME "Fyodor"
        MACOSX_BUNDLE_GUI_IDENTIFIER "com.wischner.fyodor"
        MACOSX_BUNDLE_BUNDLE_VERSION "${FYODOR_VERSION}"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "${FYODOR_VERSION}")
endif()

add_executable(test_desktop tests/ui/test_desktop.cpp tests/ui/test_resources.cpp
    tests/ui/test_selection.cpp)
target_include_directories(test_desktop PRIVATE tests/unit)
target_compile_definitions(test_desktop PRIVATE
    FIXTURES="${CMAKE_SOURCE_DIR}/tests/fixtures")
target_link_libraries(test_desktop PRIVATE fyodor_ui)
